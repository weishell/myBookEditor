/**
 * 计算图片的渲染宽度
 */
const calcImgeWidthandHeight = async url => {
  try {
    // @ts-ignore
    const { width, height } = await getImageSize(url)
    return {
      width,
      height
    }
  } catch (error) {
    return {
      width: '100%',
      height: 'auto'
    }
  }
}
/**
 * 设置新的图片节点
 * @param val 设置新的图片元素所需的参数
 */
const isValidPath = (editor: Editor, path: Path): boolean => {
  try {
    Node.get(editor, path)
    return true
  } catch (e) {
    return false
  }
}
const setImageNodes = (val: IImgParams) => {
  const { width, height, src, status, path, editor, block_id, file_sign = '',fileType = '', compressed_size = 0 } = val
  const editorUserId = store.getState().editor?.editorUserId
  // 对于非 PENDING 状态，需要先检查路径是否有效
  if (NON_INIT_FILE_STATUS.includes(status)) {
    if (!isValidPath(editor, path)) {
      // 路径已无效，可能是节点已被删除（如用户撤销操作）
      return
    }
  }
  const data = {
    block_id,
    block_type: ELTYPE_NUM.IMAGE, // 图片
    block_type_string: ELTYPE.IMAGE,
    style: {
      align: 2, // 1左对齐，2居中，3右对齐
      folded: false, // 是否折叠
      done: false, // 是否打钩完成，如 todo list
      indent: 0 // 缩进
    },
    property: {
      description: '', //图片描述
      status, // 图片加载状态：loading success fail
      offsetx: 0, //图片在x轴方向偏移量
      offsety: 0, //图片在y轴方向偏移量
      wrap_width: width, //裁剪后裁剪盒子宽度
      wrap_height: height, //裁剪后裁剪盒子高度
      url: src,
      preview_url: src,
      width,
      height,
      type: fileType, // 图片 MIME 类型，如 image/jpeg，供后端解析转发使用
      file_sign,
      editorUserId,
      compressed_size,
    },
    children: [
      {
        text: ''
        // bold: false,
        // inline_code: false,
        // italic: false,
        // strikethrough: false,
        // underline: false,
        // text_color: '' // 直接颜色即可，不用枚举
      }
    ]
  }
  // 初始化节点
  if ([FILE_STATUS.PENDING].includes(status)) {
    // Transforms.removeNodes(editor, { at: path })
    Transforms.insertNodes(editor, data, {
      at: path,
      select: false
    })
  }
  // 如果已经存在，更新属性
  else if (NON_INIT_FILE_STATUS.includes(status)) {
    setBlockNode(
      editor,
      {
        property: {
          description: '',
          status,
          offsetx: 0,
          offsety: 0,
          wrap_width: width,
          wrap_height: height,
          url: src,
          preview_url: src,
          width,
          height,
          type: fileType,
          file_sign,
          editorUserId,
          compressed_size,
        }
      },
      {
        at: path
      }
    )
  }
}

export const copyData = element => {
  try {
    // if (navigator.clipboard) {
    //   // 无法支持  e.clipboardData.setData('application/m-slate-fragment', encodeELeStr)
    //   navigator.clipboard.writeText(text).then(

    //     function (err) {
    //       console.error('Async: Could not copy text: ', err)
    //     }
    //   )
    // }
    const input = document.createElement('input')
    input.oncopy = function (e) {
      e.preventDefault()
      const eleStr = JSON.stringify(element)
      const encodeELeStr = window.btoa(encodeURIComponent(eleStr))
      // e.clipboardData.setData('text/plain', '' + element.block_id)
      e.clipboardData.setData('application/m-slate-fragment', encodeELeStr)
    }
    document.body.appendChild(input)
    input.select()
    document.execCommand('copy')
    // document.body.removeChild(input)
    if (document.body.contains(input) && input?.parentNode === document.body) {
      // 删除之前判断节点是否存在
      document.body.removeChild(input)
    }
  } catch (error) {
    console.error('copyData error,', error)
  }
}

export const imageCommand: CommandFn = (editor, key, data) => {
  // console.log(data)
  if (!isObject(data)) return
  if (!data.path) return
  let imageInput = document.getElementById(IMAGE_INPUT_DOM_ID)
  if (!imageInput) {
    // console.log('UPLOAD_ACCEPT_ALL_IMG_ACCEPT', UPLOAD_ACCEPT_ALL_IMG_ACCEPT)
    imageInput = document.createElement('input')
    imageInput.style.width = '0px'
    imageInput.style.height = '0px'
    imageInput.style.visibility = 'hidden'
    imageInput.setAttribute('type', 'file')
    imageInput.setAttribute('id', IMAGE_INPUT_DOM_ID)
    imageInput.setAttribute('accept', UPLOAD_ACCEPT_ALL_IMG_ACCEPT)
    imageInput.setAttribute('multiple', 'true')
    document.body.append(imageInput)
  }
  imageInput.onchange = e => {
    try {
      const target = e.target as HTMLInputElement
      const files = target.files
      // focusSelection(editor, data?.path)
      files && uploadImage(files, editor, data?.path)
      // document.body.removeChild(imageInput)
      if (document.body.contains(imageInput) && imageInput?.parentNode === document.body) {
        // 删除之前判断节点是否存在
        document.body.removeChild(imageInput)
      }
    } catch (error) {
      console.error('[图片]', error)
      // document.body.removeChild(imageInput)
      if (document.body.contains(imageInput) && imageInput?.parentNode === document.body) {
        // 删除之前判断节点是否存在
        document.body.removeChild(imageInput)
      }
    }
  }
  imageInput.click()
}

/**
 * 递归提取所有图片块的block_id和preview_url
 * @param {Array} data - 原始数据数组
 * @returns {Array} 包含{block_id, preview_url}的数组
 */
export const extractImageData = data => {
  // 存储结果的数组
  const result = []
  // 递归处理函数
  function traverse(item) {
    // 如果是数组，遍历每个元素
    if (Array.isArray(item)) {
      item.forEach(child => traverse(child))
      return
    }
    // 如果是对象，检查是否为图片类型
    if (typeof item === 'object' && item !== null) {
      // 图片类型的block_type为16
      if (item.block_type === 16 && item.property?.preview_url) {
        // 提取需要的字段
        result.push({
          block_id: item.block_id,
          preview_url: item.property.preview_url
        })
      }
      // 递归处理children属性（如果存在）
      if (item.children) {
        traverse(item.children)
      }
    }
  }
  // 开始遍历原始数据
  traverse(data)
  return result
}

  'image/gif',
  'image/webp',
  'image/avif'
]

const MIN_COMPRESS_SIZE = 100 * 1024
/**
 * 压缩图片
 * @param file 图片文件
 * @returns 压缩后的图片文件
 */
export const compressImage = async (
  file: File
): Promise => {
  // 跳过不处理的格式
  if (SKIP_TYPES.includes(file.type)) {
    return file
  }
  // 小图不压缩
  if (file.size < MIN_COMPRESS_SIZE) {
    return file
  }
  return new Promise((resolve) => {
    try {
      new Compressor(file, {
        quality: 0.8,
        mimeType: 'image/webp',
        convertSize: Infinity,
        maxWidth: 5120,
        maxHeight: 5120,
        success(result) {
          const compressedFile = new File(
            [result],
            file.name.replace(/.\w+$/, '.webp'),
            {
              type: 'image/webp',
              lastModified: Date.now()
            }
          )
          resolve(compressedFile)
        },
        error(err) {
          console.error('压缩图片失败', err)
          resolve(file)
        }
      })
    } catch (err) {
      console.error('压缩图片失败', err)
      resolve(file)
    }
  })
}

/**
 * 检测当前元素是否在视口内
 *
 * @param el   - 需要检测的 DOM 元素
 * @param axis - 检测轴向：
 *               'both'（默认）同时检测 X 轴和 Y 轴，即元素完全在视口内才返回 true；
 *               'y' 仅检测垂直方向（适用于纵向滚动场景，如文档评论锚点定位）；
 *               'x' 仅检测水平方向（适用于横向滚动场景）。
 */
export const isElementInViewport = (el: Element, axis: 'x' | 'y' | 'both' = 'both') => {
  try {
    const rect = el.getBoundingClientRect()
    // 垂直方向：元素顶部不超出视口上边界，底部不超出视口下边界
    const inY =
      rect.top >= 0 &&
      rect.bottom <= (window.innerHeight || document.documentElement.clientHeight)
    // 水平方向：元素左边不超出视口左边界，右边不超出视口右边界
    const inX =
      rect.left >= 0 &&
      rect.right <= (window.innerWidth || document.documentElement.clientWidth)
    if (axis === 'y') return inY
    if (axis === 'x') return inX
    return inY && inX
  } catch (error) {
    return false
  }
}

  let objectName = ''
  let uploadId = ''
  const fdsData: any = await serviceApi.uploadFileFDSId({
    docUUID,
    fileName: filename,
    fileSign: hash
  })
  if (fdsData.success) {
    const result = handleHash(fdsData)
    if (result && fdsData?.data?.uploadResult) {
      callback(100)
      return {
        ...(fdsData.data.uploadResult ?? {}),
        fileSign: fdsData.data.uploadResult?.fileSign ?? hash
      }
    }
    objectName = fdsData.data.objectName
    uploadId = fdsData.data.uploadId
  } else {
    message.error(fdsData?.errorMsg || '文件上传失败')
    return
  }
  // 初始化分片
  for (let i = 0; i < totalChunks; i++) {
    const start = i * CHUNK_SIZE
    const end = Math.min(start + CHUNK_SIZE, file.size)
    chunks.push({
      chunk: file.slice(start, end),
      chunkNumber: i + 1,
      retries: 0,
      uploaded: false,
      size: end - start
    })
  }
  // 顺序上传分片
  try {
    //  console.log('chunks', chunks)
    for (const chunkStatus of chunks) {
      let success = false
      // console.log('chunkStatus', chunkStatus)
      while (chunkStatus.retries < MAX_RETRIES && !success) {
        try {
          const formData = new FormData()
          formData.append('docUUID', docUUID as string)
          formData.append('objectName', objectName)
          formData.append('file', chunkStatus.chunk)
          // formData.append('filename', filename)
          // formData.append('fileType', fileType)
          formData.append('uploadId', uploadId as string)
          formData.append('partNumber', String(chunkStatus.chunkNumber))
          formData.append('fileSign', hash as string)
          const res = await serviceApi.uploadFileChunk(formData)
          if (res.success) {
            success = true
            chunkStatus.uploaded = true
            // 更新进度
            const uploadedCount = chunks.filter(c => c.uploaded).length
            // 这里不需要调前端的进度更新接口了，直接按切片来就好了
            callback(Math.floor((uploadedCount / totalChunks) * 100))
            fileSizeTotal += chunkStatus.size
            uploadPartResults.push(res.data)
            break
          } else {
            if (chunkStatus.retries === MAX_RETRIES) {
              message.error(res?.errorMsg ?? mes)
            }
            chunkStatus.retries++
          }
        } catch (error) {
          chunkStatus.retries++
        }
      }
      if (!success) {
        return Promise.reject({ status: FILE_STATUS.FAIL })
      }
    }
    // 所有分片上传完成后延迟请求合并
    return await new Promise((resolve, reject) => {
      setTimeout(async () => {
        try {
          const mergeRes = await serviceApi.uploadFileFinish({
            docUUID,
            filename,
            totalChunks,
            storageType: '0',
            fileType,
            fileSize: fileSizeTotal,
            uploadPartResults,
            objectName,
            uploadId,
            fileSign: hash
          })
          // console.log('mergeResmergeRes', mergeRes)
          if (mergeRes.success) {
            callback(100)
            resolve({
              ...((mergeRes?.data as any) ?? {}),
              fileSign: mergeRes?.data?.fileSign ?? hash
            })
          } else {
            reject(new Error('文件合并失败'))
          }
        } catch (error) {
          reject(new Error('最终合并请求失败'))
        }
      }, 1000)
    })
  } catch (error) {
    message.error(error.message || '文件上传失败')
    throw error
  }
}
/**
 * 上传文件处理函数
 * @param filename 文件名称
 * @param fileSize 文件大小
 * @param file 文件内容
 * @param fileType 文件类型
 * @param callback 回调 用来记录上传进度
 * @param originalFile - 原始文件，用于计算hash
 * @returns 成功返回文件url和部分信息
 */
export const handleUploadFile = async (
  filename: string,
  fileSize: number,
  file: File,
  fileType: string,
  callback = res => { },
  originalFile: File = null
) => {
  const docUUID = getSearchObjFromUrl()?.docUUID
  if (!docUUID) {
    message.error('url need docUUID')
    return
  }
  const hash = await calculateHash(originalFile || file)
  const formData = new FormData()
  formData.append('file', file)
  formData.append('docUUID', docUUID as string)
  // 云储存
  formData.append('storageType', STORAGE_TYPE.FDS)
  formData.append('filename', filename)
  formData.append('fileType', fileType)
  formData.append('fileSize', String(fileSize))
  formData.append('fileSign', hash as string)
  const mes = i18n.t('failupload')
  try {
    const res = await serviceApi.uploadFile(formData, callback)
    if (res.success) {
      return { ...((res?.data as any) ?? {}), fileSign: res?.data?.fileSign ?? hash }
    } else {
      message.error(res?.errorMsg ?? mes)
      return
    }
  } catch (error) {
    message.error(mes)
    return
  }
}
/**
 * 增强上传处理
 * @param editor 编辑器
 * @param undos 撤销记录
 * @param i index
 * @param data 文件信息
 */
const moreUpdateFileHistory = (editor: any, undos: any[], i: number, data) => {
  try {
    const { url = '', block_id, status, fileSign = '', previewUrl = '', duration, fileType = '',compressedSize = 0 } = data
    const temp = cloneDeep(undos[i])
    for (let j = 0; j < temp?.length; j++) {
      const undo = temp[j]
      if (
        undo?.type === 'insert_node' &&
        undo?.node?.block_id === block_id &&
        [FILE_STATUS.LOADING, FILE_STATUS.PENDING, FILE_STATUS.SUCCESS].includes(undo?.node?.property?.status)
      ) {
        undo.node.property.url = url
        undo.node.property.file_sign = fileSign
        // 图片的预览更新
        if (undo.node.block_type_string === ELTYPE.IMAGE) {
          undo.node.property.preview_url = url
          undo.node.property.compressed_size = compressedSize
        }
        if (undo.node.block_type_string === ELTYPE.VIDEO && previewUrl) {
          undo.node.property.preview_url = previewUrl
        }
        // 视频时长更新（秒），供后端解析转发使用
        if (undo.node.block_type_string === ELTYPE.VIDEO && Number.isFinite(duration)) {
          undo.node.property.duration = duration
        }
        // 图片 MIME 类型更新，供后端解析转发使用
        if (undo.node.block_type_string === ELTYPE.IMAGE && fileType) {
          undo.node.property.type = fileType
        }
        undo.node.property.status = status
      }
    }
    editor.history.undos[i] = temp
  } catch (error) {
    console.log(error)
  }
}

/**
 * 是否展示拉伸放大的拉伸点
 * @param element 节点
 * @returns boolean
 */
const isShowResize = (element) => {
  // 在实例模板下，如果元素来自类模板，阻止拉伸
  const isSorOperationMode = store.getState()?.editor?.isSorOperationMode
  const editMode = store.getState()?.editor?.docConfig?.editMode
  const isFromClassTemplate = element?.property?.source === SOURCE_TYPE.CLASS_TEMPLATE
  
  // 检查是否为视频、图片、画布、流程图或UML
  const isResizableElement = [
    ELTYPE.VIDEO,
    ELTYPE.IMAGE,
    ELTYPE.EXCALIDRAW,
    ELTYPE.FLOW_CHART,
    ELTYPE.UML,
  ].includes(element.block_type_string)
  // 在实例模板下，如果元素来自类模板，阻止拉伸
  if (isSorOperationMode && editMode !== TO_EDITOR_MODE.read && isFromClassTemplate && isResizableElement) {
    return false
  }
  return true
}
/**
 * 四周的四个点，拉动可缩放图片
 */
export const onMouseDown = (
  e: React.MouseEvent<HTMLDivElement, MouseEvent> | React.TouchEvent,
  props
) => {
  const { blockId, path, propref, callback } = props
  const config = props.element
  let centerPoint = false
  if ([ELTYPE.FLOW_CHART, ELTYPE.UML, ELTYPE.EXCALIDRAW].includes(config.block_type_string)) {
    centerPoint = true
  }
  const tag = canDrag(props.editor, path)
  if (!tag) return
  if (!isShowResize(config)) return
  try {
    // 支持鼠标和触摸事件
    let clientX: number, clientY: number
    if ('touches' in e) {
      // 触摸事件
      clientX = e.touches[0].clientX
      clientY = e.touches[0].clientY
    } else {
      // 鼠标事件
      clientX = e.clientX
      clientY = e.clientY
    }
    const resider = e.target as HTMLDivElement
    const element = propref.current
    let temWidth = 0
    let oldLen = props.editor.history.undos.length ?? 0
    e.stopPropagation()
    e.preventDefault() // 阻止默认行为，包括移动端的滚动和缩放
    if (resider instanceof HTMLDivElement && resider.dataset.type && element && config) {
      const startX = clientX
      const startY = clientY
      const startWidth = element.offsetWidth
      const startHeight = element.offsetHeight
      const ratio = startWidth / startHeight
      const type = resider.dataset.type
      if (!path) return
      // 统一处理鼠标和触摸事件
      const handleMove = (moveEvent: MouseEvent | TouchEvent) => {
        try {
          let currentX: number, currentY: number
          if ('touches' in moveEvent) {
            currentX = moveEvent.touches[0].clientX
            currentY = moveEvent.touches[0].clientY
            // 在移动过程中持续阻止默认行为
            moveEvent.preventDefault()
          } else {
            currentX = moveEvent.clientX
            currentY = moveEvent.clientY
          }
          const diffX = currentX - startX
          const diffY = currentY - startY
          let width = startWidth
          let height = startHeight
          switch (type) {
            case 'lt':
              width -= diffX
              height -= diffY
              break
            case 'rt':
              width += diffX
              height -= diffY
              break
            case 'lb':
              width -= diffX
              height += diffY
              break
            case 'rb':
              width += diffX
              height += diffY
              break
            case 'ct':
              height -= diffY
              break
            case 'cb':
              height += diffY
              break
            case 'cl':
              width -= diffX
              break
            case 'cr':
              width += diffX
              break
          }
          if (centerPoint) {
            if (width <= 150) {
              width = 150
            }
            if (height <= 150) {
              height = 150
            }
          } else {
            if (width / height > ratio) {
              height = width / ratio
            } else {
              width = height * ratio
            }
            // if (width >= 50) {
            temWidth = width
            // }
          }
          // temWidth = Math.max(width, 50)
          if (centerPoint && (width < 150 || height < 150)) {
            return
          }
          if (width > startWidth) {
            // 当前是因为分栏表格等特殊情况造成宽高小于50，现在需要变大，这里需要给过
            // console.log(width, startWidth)
          } else if (width < 50 || height < 50) {
            // TODO 优化：存在快速往里拖拽，虽然model数据不变，但是页面图片会略微缩小，需要优化
            removeEventListeners()
            return
          }
          // 因为分栏等情况会造成极度的收缩宽高比
          if (props.element.block_type_string === ELTYPE.IMAGE) {
            const zoom = temWidth / config.property.wrap_width
            callback(width, zoom)
          } else {
            callback(width, height)
          }
        } catch (error) {
          console.error('[Drag组件]事件报错:', error)
        }
      }
      const { isMobile } = checkDevice()
      const handleUp = () => {
        try {
          // TODO ↓↓↓↓↓待验证：如果后续存在不合理，需要重新调整↓↓↓↓↓
          const undoArr = props.editor.history.undos
          const newLen = undoArr.length ?? 0
          if (newLen > oldLen) {
            // 直接操作是否会影响协同 yjs？？
            props.editor.history.undos = undoArr.slice(0, oldLen + 1)
          } else {
            if (props.element.block_type_string === ELTYPE.IMAGE) {
              const zoom = temWidth / props.element.property.wrap_width
              // 鼠标点击后没拖拽
              if (zoom > 0) {
                callback(temWidth, zoom)
              }
            }
          }
        } catch (error) {}
        removeEventListeners()
        if (isMobile) {
          // 拖拽结束后恢复页面滚动
          restorePageScroll()
        }
      }
      // 阻止页面滚动的函数
      const preventPageScroll = () => {
        const rootNode = document.querySelector('#root')
        // 阻止页面滚动
        if (rootNode && rootNode instanceof HTMLElement) {
          rootNode.style.overflowY = 'hidden'
          rootNode.style.touchAction = 'none' // 防止触摸动作
          rootNode.style.userSelect = 'none' // 防止文字选择
        }
      }
      // 恢复页面滚动的函数
      const restorePageScroll = () => {
        const rootNode = document.querySelector('#root')
        // 阻止页面滚动
        if (rootNode && rootNode instanceof HTMLElement) {
          rootNode.style.overflowY = 'auto'
          rootNode.style.touchAction = ''
          rootNode.style.userSelect = ''
        }
      }
      const throttledHandleMove = throttle(handleMove, 16) // 约60fps
      // 添加事件监听器
      const addEventListeners = () => {
        if (isMobile) {
          preventPageScroll() // 开始拖拽时阻止页面滚动
        }
        if ('touches' in e) {
          // 触摸设备
          document.addEventListener('touchmove', throttledHandleMove, { passive: false })
          document.addEventListener('touchend', handleUp, { passive: false })
          document.addEventListener('touchcancel', handleUp, { passive: false })
        } else {
          // 鼠标设备
          document.addEventListener('pointermove', throttledHandleMove)
          document.addEventListener('pointerup', handleUp)
        }
      }
      // 移除事件监听器
      const removeEventListeners = () => {
        document.removeEventListener('touchmove', throttledHandleMove)
        document.removeEventListener('touchend', handleUp)
        document.removeEventListener('touchcancel', handleUp)
        document.removeEventListener('pointermove', throttledHandleMove)
        document.removeEventListener('pointerup', handleUp)
      }
      addEventListeners()
    }
  } catch (error) {
    console.error('error', error)
  }
}

 * 判断类型，是走四个点位不形变拖动，还是走八个点位的支持形变拖动
 */

            data-type={pointType}
            className={cx((props.selected || props.showFourPoint) && style['app-drag-resider'])}
            onMouseDown={event => {

            onTouchStart={event => {

          >
        ))}
      </>
    )
  }
  // 其他元素，只支持四角的移动
  return (
    <>
      {['lt', 'rt', 'lb', 'rb'].map(pointType => (
        <div
          key={pointType}
          data-type={pointType}
          className={cx((props.selected || props.showFourPoint) && style['app-drag-resider'])}
          onMouseDown={event => {
            onMouseDown(event, props)
          }}
          onTouchStart={event => {
            onMouseDown(event, props)
          }}
        >
      ))}
    </>
  )
}
